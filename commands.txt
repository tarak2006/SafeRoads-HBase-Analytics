# ------------------------------------------------------------------------------
# 4. HBASE FILTERS (EXACTLY 6 FILTERS) (4 Marks)
# Script File: 04_filter_queries.hbase
# ------------------------------------------------------------------------------
# Filter 1: PrefixFilter (Ohio state crashes)
scan 'saferoads_accidents', {FILTER => "PrefixFilter('OH#')", LIMIT => 5}

# Filter 2: SingleColumnValueFilter Substring (Rain conditions)
scan 'saferoads_accidents', {FILTER => "SingleColumnValueFilter('env', 'weather', =, 'substring:Rain')", LIMIT => 5}

# Filter 3: SingleColumnValueFilter Binary (Highway Junctions)
scan 'saferoads_accidents', {FILTER => "SingleColumnValueFilter('hazard', 'junction', =, 'binary:1')", LIMIT => 5}

# Filter 4: SingleColumnValueFilter Binary (Signalized Intersections)
scan 'saferoads_accidents', {FILTER => "SingleColumnValueFilter('hazard', 'traffic_signal', =, 'binary:1')", LIMIT => 5}

# Filter 5: SingleColumnValueFilter Comparison >= (Duration >= 60 min)
scan 'saferoads_accidents', {COLUMNS => ['loc:city', 'loc:state', 'time:duration_min'], FILTER => "SingleColumnValueFilter('time', 'duration_min', >=, 'binary:60.0')", LIMIT => 5}

# Filter 6: Compound FilterList MUST_PASS_ALL (AND) (Severity >= 3 AND Junction = 1)
scan 'saferoads_accidents', {FILTER => "(SingleColumnValueFilter('hazard', 'severity', >=, 'binary:3') AND SingleColumnValueFilter('hazard', 'junction', =, 'binary:1'))", LIMIT => 5}
